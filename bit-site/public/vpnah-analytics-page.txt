<script>
(function () {
  'use strict';
  var params = new URLSearchParams(location.search);
  var days = Number(params.get('days'));
  if ([1, 7, 30].indexOf(days) < 0) days = 7;
  var controller = null, requestNumber = 0;
  var names = { mobile: '手机 / 平板', desktop: 'PC', unknown: '未识别设备' };
  var channels = [['app-store', 'App Store'], ['google-play', 'Google Play'], ['testflight', 'TestFlight'], ['android-apk', '安卓 APK']];
  function element(id) { return document.getElementById(id); }
  function number(value) { return Number(value || 0).toLocaleString('zh-CN'); }
  function table(id, values) {
    var body = element(id);
    body.replaceChildren();
    values.forEach(function (values) {
      var row = document.createElement('tr');
      values.forEach(function (value, index) {
        var cell = document.createElement('td');
        cell.textContent = String(value);
        if (index === values.length - 1) cell.className = 'num-primary';
        row.appendChild(cell);
      });
      body.appendChild(row);
    });
  }
  function render(data) {
    if (data.scope !== 'VPNAH' || !Array.isArray(data.pages) || !Array.isArray(data.devices) || !Array.isArray(data.downloads)) throw new Error('返回的数据格式不正确，请刷新重试');
    var landing = data.pages.find(function (row) { return row.path === '/VPNAH'; });
    var guide = data.pages.find(function (row) { return row.path === '/VPNAH/tutorial'; });
    if (!landing || !guide) throw new Error('页面统计暂不完整，请刷新重试');
    element('landingVisitors').textContent = number(landing.visitors);
    element('landingViews').textContent = number(landing.pageviews);
    element('tutorialClicks').textContent = number(landing.tutorial_clicks);
    element('downloadClicks').textContent = number(guide.download_clicks);
    table('landingRows', data.devices.filter(function (row) { return row.path === '/VPNAH'; }).map(function (row) {
      return [names[row.device] || names.unknown, number(row.pageviews), number(row.visitors), number(row.tutorial_clicks), row.pageviews ? (row.tutorial_clicks / row.pageviews * 100).toFixed(1) + '%' : '—', number(row.signup_clicks)];
    }));
    table('guideRows', data.devices.filter(function (row) { return row.path === '/VPNAH/tutorial'; }).map(function (row) {
      return [names[row.device] || names.unknown, number(row.pageviews), number(row.visitors), number(row.download_clicks)];
    }));
    var counts = {};
    channels.concat([['unknown', '其他 / 未识别渠道']]).forEach(function (channel) { counts[channel[0]] = { mobile: 0, desktop: 0, unknown: 0 }; });
    data.downloads.filter(function (row) { return row.path === '/VPNAH/tutorial'; }).forEach(function (row) {
      var channel = channels.some(function (channel) { return channel[0] === row.channel; }) ? row.channel : 'unknown';
      var device = row.device === 'mobile' || row.device === 'desktop' ? row.device : 'unknown';
      counts[channel][device] += Number(row.clicks || 0);
    });
    var visibleChannels = channels.slice();
    if (Object.values(counts.unknown).some(function (value) { return value > 0; })) visibleChannels.push(['unknown', '其他 / 未识别渠道']);
    table('downloadRows', visibleChannels.map(function (channel) {
      var row = counts[channel[0]];
      return [channel[1], number(row.mobile), number(row.desktop), number(row.unknown), number(row.mobile + row.desktop + row.unknown)];
    }));
    element('empty').hidden = data.pages.some(function (row) { return row.pageviews || row.tutorial_clicks || row.signup_clicks || row.download_clicks; });
    element('report').hidden = false;
    element('status').textContent = '最近 ' + (data.days === 1 ? '24 小时' : data.days + ' 天') + ' · 更新于 ' + new Date(data.generated_at).toLocaleString('zh-CN', { timeZone: 'Asia/Shanghai', hour12: false }) + '（北京时间）';
  }
  async function load() {
    var current = ++requestNumber;
    if (controller) controller.abort();
    controller = new AbortController();
    element('refresh').disabled = true;
    element('error').hidden = true;
    element('report').hidden = true;
    element('status').textContent = '正在读取数据…';
    document.querySelectorAll('[data-days]').forEach(function (button) { button.setAttribute('aria-pressed', String(Number(button.dataset.days) === days)); });
    try {
      var response = await fetch('/VPNAH/analytics/data?days=' + days, { credentials: 'same-origin', cache: 'no-store', signal: controller.signal });
      if (!response.ok) throw new Error(response.status === 401 ? '登录已失效，请刷新页面重新登录' : '数据暂时无法读取，请稍后点击刷新数据');
      var data = await response.json();
      if (current !== requestNumber) return;
      render(data);
    } catch (error) {
      if (current !== requestNumber || error.name === 'AbortError') return;
      element('error').textContent = error.message || '读取失败，请稍后重试';
      element('error').hidden = false;
      element('report').hidden = true;
      element('status').textContent = '本次未取得数据，不显示为 0。';
    } finally {
      if (current === requestNumber) element('refresh').disabled = false;
    }
  }
  document.querySelectorAll('[data-days]').forEach(function (button) {
    button.addEventListener('click', function () {
      days = Number(button.dataset.days);
      var url = new URL(location.href); url.searchParams.set('days', String(days)); history.replaceState(null, '', url);
      load();
    });
  });
  element('refresh').addEventListener('click', load);
  load();
})();
</script>
